const APP_SPLIT_REGEX_SOURCE =
  '(?:[\\u2700-\\u27bf]|(?:\\ud83c[\\udde6-\\uddff]){2}|[\\ud800-\\udbff][\\udc00-\\udfff]|[\\u0023-\\u0039]\\ufe0f?\\u20e3|\\u3299|\\u3297|\\u303d|\\u3030|\\u24c2|\\ud83c[\\udd70-\\udd71]|\\ud83c[\\udd7e-\\udd7f]|\\ud83c\\udd8e|\\ud83c[\\udd91-\\udd9a]|\\ud83c[\\udde6-\\uddff]|[\\ud83c[\\ude01-\\ude02]|\\ud83c\\ude1a|\\ud83c\\ude2f|[\\ud83c[\\ude32-\\ude3a]|[\\ud83c[\\ude50-\\ude51]|\\u203c|\\u2049|[\\u25aa-\\u25ab]|\\u25b6|\\u25c0|[\\u25fb-\\u25fe]|\\u00a9|\\u00ae|\\u2122|\\u2139|\\ud83c\\udc04|[\\u2600-\\u26FF]|\\u2b05|\\u2b06|\\u2b07|\\u2b1b|\\u2b1c|\\u2b50|\\u2b55|\\u231a|\\u231b|\\u2328|\\u23cf|[\\u23e9-\\u23f3]|[\\u23f8-\\u23fa]|\\ud83c\\udccf|\\u2934|\\u2935|[\\u2190-\\u21ff])';

const SUPPORTED_TYPES = ['Title', 'Text', 'Image'];

function findSplittingChars(text) {
    const re = new RegExp(APP_SPLIT_REGEX_SOURCE, 'g');
    const found = [];
    let m;
    while ((m = re.exec(text))) found.push(m[0]);
    return found.filter((v, i) => found.indexOf(v) === i);
}

function describeChar(ch) {
    return ch === '[' ? 'nawias otwierający "["' : `emoji lub symbol "${ch}"`;
}

function snippet(s, max) {
    const clean = String(s).replace(/\n/g, ' ').trim();
    return clean.length > (max || 90) ? clean.slice(0, max || 90) + '...' : clean;
}

function checkTextBlock(text, label, errors, warnings) {
    if (!text || text.trim().length === 0) {
        warnings.push(`${label}: blok jest pusty i nic nie wyświetli.`);
        return;
    }

    // 1. Znak rozcinający wewnątrz <b> lub <a>: tagi pokażą się dosłownie.
    [
        {name: 'pogrubienie <b>', re: /<b>(.+?)<\/b>/g},
        {name: 'link <a>', re: /<a(.+?)<\/a>/g},
    ].forEach(({name, re}) => {
        let m;
        while ((m = re.exec(text))) {
            const bad = findSplittingChars(m[0]);
            if (bad.length > 0) {
                errors.push(`${label}: ${name} NIE ZADZIAŁA. W aplikacji zamiast formatowania pokażą się dosłownie tagi. Powód: w środku jest ${bad.map(describeChar).join(', ')}, a aplikacja przerywa formatowanie na tym znaku. Fragment: \`${snippet(m[0])}\``);
            }
        }
    });

    // 2. Nawias "[" gdziekolwiek w tekście: aplikacja go nie obsługuje.
    const brackets = (text.match(/\[/g) || []).length;
    if (brackets > 0) {
        warnings.push(`${label}: ${brackets}x nawias otwierający "[". Aplikacja NIE OBSŁUGUJE tego znaku w blokach Text, wyświetli go inną czcionką niż reszta zdania (wygląda jak pogrubiony). Zamykający "]" jest bezpieczny. Zalecenie: użyj nawiasów okrągłych ( ).`);
    }

    // 3. Tagi inne niż <b>, </b>, <a ...>, </a>: apka wypisze je dosłownie.
    const tagRe = /<[^>]*>/g;
    let t;
    while ((t = tagRe.exec(text))) {
        const tag = t[0];
        const ok = /^<b>$/.test(tag) || /^<\/b>$/.test(tag) || /^<a\s[^>]*>$/.test(tag) || /^<\/a>$/.test(tag);
        if (!ok) {
            errors.push(`${label}: tag \`${tag}\` NIE JEST OBSŁUGIWANY i wyświetli się dosłownie. Aplikacja rozumie wyłącznie \`<b>\`, \`</b>\`, \`<a href="...">\`, \`</a>\` (małymi literami, bez spacji w środku tagu).`);
        }
    }

    // 4. Niesparowane tagi.
    const openB = (text.match(/<b>/g) || []).length;
    const closeB = (text.match(/<\/b>/g) || []).length;
    if (openB !== closeB) {
        errors.push(`${label}: niesparowane pogrubienie. Znaleziono \`<b>\` x${openB}, a \`</b>\` x${closeB}. Każde otwarcie musi mieć zamknięcie.`);
    }
    const openA = (text.match(/<a[\s>]/g) || []).length;
    const closeA = (text.match(/<\/a>/g) || []).length;
    if (openA !== closeA) {
        errors.push(`${label}: niesparowany link. Znaleziono \`<a>\` x${openA}, a \`</a>\` x${closeA}.`);
    }

    // 5. Tag rozjeżdżający się na dwie linie: aplikacja tego nie sklei.
    const brokenLines = text.split('\n').filter(line => {
        const o = (line.match(/<b>/g) || []).length;
        const c = (line.match(/<\/b>/g) || []).length;
        const oa = (line.match(/<a[\s>]/g) || []).length;
        const ca = (line.match(/<\/a>/g) || []).length;
        return o !== c || oa !== ca;
    });
    if (brokenLines.length > 0 && openB === closeB && openA === closeA) {
        errors.push(`${label}: formatowanie nie może przechodzić przez nową linię. Otwórz i zamknij tag w obrębie jednego akapitu. Linia: \`${snippet(brokenLines[0], 80)}\``);
    }

    // 6. Zagnieżdżone tagi: aplikacja obsługuje tylko jeden poziom.
    let n;
    const nestedA = /<a(.+?)<\/a>/g;
    while ((n = nestedA.exec(text))) {
        if (n[0].includes('<b>')) {
            errors.push(`${label}: pogrubienie w środku linku NIE DZIAŁA. Aplikacja przetworzy najpierw \`<b>\`, a tagi linku zostaną wypisane dosłownie i link będzie nieklikalny. Fragment: \`${snippet(n[0])}\``);
        }
    }
    const nestedB = /<b>(.+?)<\/b>/g;
    while ((n = nestedB.exec(text))) {
        if (/<a[\s>]/.test(n[1])) {
            errors.push(`${label}: link w środku pogrubienia NIE DZIAŁA. Aplikacja wypisze tagi linku dosłownie wewnątrz pogrubionego tekstu. Fragment: \`${snippet(n[0])}\``);
        }
    }

    // 7. Pusty tag: aplikacja wymaga co najmniej jednego znaku w środku.
    if (/<b>\s*<\/b>/.test(text)) {
        errors.push(`${label}: puste pogrubienie \`<b></b>\`. Aplikacja pokaże same tagi. Usuń je.`);
    }

    // 8. Link: adres musi być PIERWSZĄ wartością w podwójnych cudzysłowach.
    const linkRe = /<a(.+?)<\/a>/g;
    let l;
    while ((l = linkRe.exec(text))) {
        const whole = l[0];
        const firstQuoted = whole.match(/"(.+?)"/);
        if (!firstQuoted) {
            const hasApostrophes = /'/.test(whole.split('>')[0]);
            errors.push(`${label}: ${hasApostrophes ? 'adres podany w apostrofach nie zadziała, aplikacja szuka podwójnych cudzysłowów' : 'link nie ma adresu w podwójnych cudzysłowach, więc kliknięcie nic nie zrobi'}. Wymagany zapis: \`<a href="https://...">tekst</a>\`. Fragment: \`${snippet(whole)}\``);
        } else {
            const before = whole.slice(0, firstQuoted.index);
            if (!/href=$/.test(before)) {
                errors.push(`${label}: \`href\` musi być PIERWSZYM atrybutem linku. Aplikacja bierze pierwszą wartość w cudzysłowach i tutaj otworzyłaby "${firstQuoted[1]}" zamiast adresu. Fragment: \`${snippet(whole)}\``);
            } else if (!/^https?:\/\//i.test(firstQuoted[1])) {
                warnings.push(`${label}: adres linku "${firstQuoted[1]}" nie zaczyna się od http:// ani https://, kliknięcie może pokazać błąd.`);
            }
        }
    }

    // 9. Encje HTML: aplikacja ich nie dekoduje.
    const entities = text.match(/&[a-z]+;|&#\d+;/gi);
    if (entities) {
        errors.push(`${label}: encje HTML ${entities.filter((v, i) => entities.indexOf(v) === i).join(', ')} wyświetlą się dosłownie. Wpisz znak wprost, np. "&" zamiast "&amp;".`);
    }
}

function checkTitleBlock(text, label, errors, warnings) {
    if (!text || text.trim().length === 0) {
        warnings.push(`${label}: tytuł jest pusty.`);
        return;
    }
    if (/<[^>]*>/.test(text)) {
        errors.push(`${label}: tytuły w ogóle nie obsługują tagów HTML, wyświetlą się dosłownie. Przenieś formatowany fragment do bloku Text.`);
    }
    if (text.includes('\n')) {
        warnings.push(`${label}: tytuł zawiera złamanie linii, w aplikacji tytuł to jeden akapit.`);
    }
}

function checkImageBlock(item, label, errors, warnings) {
    const meta = item.metadata;
    if (!meta) {
        errors.push(`${label}: brak metadanych, zdjęcie się nie wyświetli.`);
        return;
    }
    const path = meta.imageRelativePath;
    if (!path || String(path).trim() === '') {
        errors.push(`${label}: brak ścieżki do pliku, zdjęcie się nie wyświetli.`);
    } else {
        if (/\s/.test(path)) {
            warnings.push(`${label}: ścieżka pliku zawiera spacje: \`${path}\`. Adres nie jest kodowany, na części urządzeń zdjęcie może się nie załadować. Zalecenie: nazwy plików bez spacji.`);
        }
        if (/[^\x20-\x7E]/.test(path)) {
            warnings.push(`${label}: ścieżka pliku zawiera polskie znaki lub znaki specjalne: \`${path}\`. To samo ryzyko co przy spacjach.`);
        }
    }
    const ratio = meta.heightToWidthRatio;
    if (ratio === null || ratio === undefined || ratio <= 0) {
        errors.push(`${label}: brak lub zerowe \`heightToWidthRatio\`, aplikacja wyliczy wysokość 0 i zdjęcie będzie niewidoczne.`);
    } else if (ratio >= 5) {
        errors.push(`${label}: \`heightToWidthRatio\` = ${ratio} jest poza dozwolonym zakresem (0, 5), zapis zostanie odrzucony przez backend.`);
    }
}

function convertAdminFancyContentToMarkdown(content) {
    if (content == null || content == undefined || content.length === 0 || Object.keys(content).length === 0) {
        return 'Brak treści';
    }

    const errors = [];
    const warnings = [];

    const body = content.map((item, index) => {
        const label = `Blok ${index + 1} (${item.type})`;

        if (!SUPPORTED_TYPES.includes(item.type)) {
            errors.push(`Blok ${index + 1}: nieznany typ "${item.type}". Aplikacja pominie ten blok i nie wyświetli nic. Dozwolone typy: ${SUPPORTED_TYPES.join(', ')}.`);
        } else if (item.type === 'Text') {
            checkTextBlock(item.content || '', label, errors, warnings);
        } else if (item.type === 'Title') {
            checkTitleBlock(item.content || '', label, errors, warnings);
        } else if (item.type === 'Image') {
            checkImageBlock(item, label, errors, warnings);
        }

        switch (item.type) {
            case 'Title':
                return `#### ${item.content}`;
            case 'Text':
                return item.content;
            case 'Image':
                return item.metadata?.imageRelativePath ? `<img src="https://blobs.codzienniefitapp.com/public/${item.metadata.imageRelativePath}"/>` : '';
            default:
                return item.content || '';
        }
    }).filter(text => text.length > 0).join('\n\n');

    const banner = [];
    if (errors.length > 0) {
        banner.push(`> 🛑 **BŁĄD FORMATOWANIA (${errors.length})**`);
        errors.forEach(e => banner.push(`> - ${e}`));
    }
    if (warnings.length > 0) {
        if (banner.length > 0) banner.push('>');
        banner.push(`> ⚠️ **UWAGA (${warnings.length})**`);
        warnings.forEach(w => banner.push(`> - ${w}`));
    }
    if (banner.length === 0) return body;

    banner.push('>');
    banner.push('> **Co obsługuje aplikacja:**');
    banner.push('> - Bloki: `Title`, `Text`, `Image`. Inne typy są pomijane.');
    banner.push('> - W bloku `Text` tylko `<b>tekst</b>` oraz `<a href="https://...">tekst</a>`, bez zagnieżdżania, w obrębie jednej linii.');
    banner.push('> - Znak `[` nie jest obsługiwany nigdzie w bloku `Text`, nawet poza pogrubieniem. Zamień `[AUDIO]` na `(AUDIO)`.');
    banner.push('> - Emoji trzymaj poza `<b>...</b>`, np. `<b>Nowy plan</b> 🎧` zamiast `<b>Nowy plan 🎧</b>`.');
    banner.push('> - W blokach `Title` nawiasy i emoji są bezpieczne, tytuły nie są parsowane.');

    return banner.join('\n') + '\n\n---\n\n' + body;
}

### JS QUERY moveDayForCurrentPath  (additionalScope params: ['levelId', 'dayIndex', 'direction'])
####################################################################################################
function moveDay(pathsData, targetLevelId, dayIndex, direction) {
  console.log(`[moveDay] Starting: dayIndex ${dayIndex}, direction ${direction}, targetLevelId ${targetLevelId}`);
  console.log('[moveDay] Initial pathsData:', JSON.stringify(pathsData, null, 2));
  
  // Create a deep copy of pathsData
  const newPathsData = JSON.parse(JSON.stringify(pathsData));
  console.log('[moveDay] Deep copy created');
  
  if (!newPathsData[targetLevelId]) {
    console.error(`[moveDay] Error: Level ${targetLevelId} not found in the challenge data.`);
    return newPathsData;
  }
  
  const level = newPathsData[targetLevelId];
  console.log('[moveDay] Target level:', JSON.stringify(level, null, 2));
  
  const currentIndex = level.days.findIndex(day => day.dayIndex === dayIndex);
  if (currentIndex === -1) {
    console.error(`[moveDay] Error: Day with index ${dayIndex} not found in level ${targetLevelId}.`);
    return newPathsData;
  }
  
  const newIndex = currentIndex + direction;
  console.log(`[moveDay] Current index: ${currentIndex}, New index: ${newIndex}`);
  
  if (newIndex < 0 || newIndex >= level.days.length) {
    console.error(`[moveDay] Error: Cannot move day outside of valid range. New index: ${newIndex}, Days length: ${level.days.length}`);
    return newPathsData;
  }
  
  // Swap the days
  [level.days[currentIndex], level.days[newIndex]] = [level.days[newIndex], level.days[currentIndex]];
  console.log('[moveDay] Days swapped');
  
  // Reindex all days
  level.days = level.days.map((day, index) => ({
    ...day,
    dayIndex: index + 1
  }));
  console.log('[moveDay] Days reindexed');
  
  console.log('[moveDay] Final level state:', JSON.stringify(level, null, 2));
  console.log('[moveDay] Final pathsData:', JSON.stringify(newPathsData, null, 2));
  return newPathsData;
}

// The handleMoveDay function remains the same
function handleMoveDay(levelId, dayIndex, direction) {
  console.log(`[handleMoveDay] Starting: levelId ${levelId}, dayIndex ${dayIndex}, direction ${direction}`);
  
  let currentData = challengePathsBeingEdited.value;
  console.log('[handleMoveDay] Current state:', JSON.stringify(currentData, null, 2));
  
  let newData = moveDay(currentData, levelId, dayIndex, direction);
  console.log('[handleMoveDay] Move operation completed');
  
  challengePathsBeingEdited.setValue(newData);
  console.log('[handleMoveDay] State updated with setValue');
  
  console.log('[handleMoveDay] New state after setValue:', JSON.stringify(challengePathsBeingEdited.value, null, 2));
  
  console.log('[handleMoveDay] Completed. UI update may be needed.');
}

### EVENT SCRIPT copyMessageImageJson [ButtonWidget2] on "click"
####################################################################################################
if (messageImagesByName.value[item.path]){
  utils.copyToClipboard(messageImagesByName.value[item.path])

return

}

(async () => {
imageNameBeingPreparedForMessages.setValue(item.path)
    try {
        // Step 0: Block UI - Show loading indicator
        utils.showNotification({ title: "Processing", description: "Fetching and processing image..." });

        console.log("Step 1: Fetching image with path:", item.path);
        
        // 1. Fetch Image from Retool Query
        const response = await fetchImage.trigger({ additionalScope: { path: item.path } });
        console.log("Step 2: Image fetched successfully.");

        // Debug: Log response keys
        console.log("Debug: Response structure keys:", Object.keys(response));

        // 2. Validate Response
        if (!response || !response.base64Data) {
            throw new Error("No base64Data received from fetchImage.");
        }
        console.log("Step 3: Base64 data found (not logged for performance).");

        // 3. Convert Base64 to Image
        const img = new Image();
        img.src = `data:image/png;base64,${response.base64Data}`;
        console.log("Step 4: Image source set.");

        // 4. Get Image Dimensions
        const ratio = await new Promise((resolve, reject) => {
            img.onload = () => {
                console.log("Step 5: Image loaded successfully.");
                resolve(img.naturalHeight / img.naturalWidth);
            };
            img.onerror = (err) => {
                console.error("Step 5 Error: Image failed to load.", err);
                reject(new Error("Could not load image from Base64."));
            };
        });

        console.log("Step 6: Height-to-width ratio calculated:", ratio);

        // 5. Generate JSON Output
        const jsonData = JSON.stringify({
            "type": "Image",
            "content": "",
            "metadata": {
                "imageRelativePath": item.path,
                "heightToWidthRatio": ratio
            }
        }, null, 2);
        console.log("Step 7: JSON generated successfully.");

        // 6. Store JSON Data in messageImagesByName using setValue
        messageImagesByName.setValue({
            ...messageImagesByName.value,
            [item.path]: jsonData
        });
        console.log("Step 8: JSON stored in messageImagesByName.");

        // 7. Show success notification
        utils.showNotification({ title: "Stored", description: "Image metadata stored successfully." });

    } catch (error) {
        console.error("Error occurred:", error);
        utils.showNotification({ title: "Error", description: error.message || "Could not process the image." });
    } finally {
imageNameBeingPreparedForMessages.setValue("")
    }
})();

### TABLE BadgesTable column 51798 (key=id, label=Info techniczne) value expression
####################################################################################################
{{ 
  currentSourceRow.type == "milestone" 
    ? currentSourceRow.milestoneCount + "x" + " (" + currentSourceRow.repeatCategory + ")" 
    : currentSourceRow.type == "normal" 
      ? (() => {
          const w = currentSourceRow.assignedToWorkouts?.length || 0;
          const c = currentSourceRow.assignedToChallenges?.length || 0;
          const wStr = w ? w + " " + (w === 1 ? "treningu" : "treningów") : "";
          const cStr = c ? c + " " + (c === 1 ? "wyzwania" : "wyzwań") : "";
          if (!w && !c) return "Nieprzypisana";
          return "Przypisana do " + [wStr, cStr].filter(Boolean).join(" i ");
        })()
      : "" 
}}
